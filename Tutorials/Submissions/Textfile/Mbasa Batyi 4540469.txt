SET SERVEROUTPUT ON;
ACCEPT emp_id_input NUMBER PROMPT 'Enter the employee id: ';

DECLARE
    iEmp_id        HR.employees.employee_id%TYPE := &emp_id_input;
    sFirst_name    HR.employees.first_name%TYPE;
    sLast_name     HR.employees.last_name%TYPE;
    sJob_id        HR.employees.job_id%TYPE;
    sRow           VARCHAR2(200);
    sBorder        VARCHAR2(200);

BEGIN
    SELECT first_name,
           last_name,
           job_id
            INTO   sFirst_name,
                   sLast_name,
                   sJob_id
    FROM   HR.employees
    WHERE  employee_id = iEmp_id;

    sRow := '| ' || sLast_name || ' | ' || sFirst_name || ' | ' || sJob_id || ' |';

    sBorder := '+';
    FOR i IN 1 .. LENGTH(sRow) - 2
        LOOP
            sBorder := sBorder || '-';
        END LOOP;
    sBorder := sBorder || '+';

    FOR i IN 1 .. LENGTH(sBorder)
        LOOP
            DBMS_OUTPUT.PUT(SUBSTR(sBorder, i, 1));
        END LOOP;
        
    DBMS_OUTPUT.NEW_LINE;
    DBMS_OUTPUT.PUT_LINE(sRow);

    FOR i IN 1 .. LENGTH(sBorder)
        LOOP
            DBMS_OUTPUT.PUT(SUBSTR(sBorder, i, 1));
        END LOOP;
    DBMS_OUTPUT.NEW_LINE;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('No employee found with ID ' || iEmp_id);

END;
/

SET SERVEROUTPUT ON;
DECLARE
    CURSOR emp_cursor IS
        SELECT e.employee_id,
               e.first_name,
               e.last_name,
               e.salary,
               j.job_title
        FROM   HR.employees e
        JOIN   HR.jobs j ON e.job_id = j.job_id
        WHERE  e.department_id = 60
        ORDER BY e.salary DESC;

    iEmployee_id   HR.employees.employee_id%TYPE;
    sFirst_name    HR.employees.first_name%TYPE;
    sLast_name     HR.employees.last_name%TYPE;
    iSalary        HR.employees.salary%TYPE;
    sJob_title      HR.jobs.job_title%TYPE;

    iCount NUMBER := 0;

BEGIN
    OPEN emp_cursor;
        LOOP
            FETCH emp_cursor
                INTO iEmployee_id,
                     sFirst_name,
                     sLast_name,
                     iSalary,
                     sJob_title;
            EXIT WHEN emp_cursor%NOTFOUND;

            iCount := iCount + 1;

            IF iSalary < 5000 THEN
                DBMS_OUTPUT.PUT_LINE(sFirst_name || ' ' || sLast_name ||
                    ' works as ' || sJob_title || ' with salary ' || iSalary ||
                    ' (Standard Earner)');
            ELSE
                DBMS_OUTPUT.PUT_LINE(sFirst_name || ' ' || sLast_name ||
                    ' works as ' || sJob_title || ' with salary ' || iSalary ||
                    ' (Higher Earner)');
            END IF;
        END LOOP;
    CLOSE emp_cursor;
END;
/

SET SERVEROUTPUT ON;

CREATE OR REPLACE PROCEDURE dept_salary_report (proc_dept_id IN HR.departments.department_id%TYPE)
IS
    sDept_name   HR.departments.department_name%TYPE;
    nAvg_salary  NUMBER;
    sBorder      VARCHAR2(60) := RPAD('-', 60, '-');

    CURSOR emp_cursor IS
        SELECT first_name,
               last_name,
               job_id,
               salary
        FROM   HR.employees
        WHERE  department_id = proc_dept_id
        ORDER BY salary DESC, employee_id;

    sTag         VARCHAR2(20);
BEGIN
    SELECT department_name
          INTO   sDept_name
    FROM   HR.departments
    WHERE  department_id = proc_dept_id;

    SELECT AVG(salary)
           INTO   nAvg_salary
    FROM   HR.employees
    WHERE  department_id = proc_dept_id;

    DBMS_OUTPUT.PUT_LINE('Department: ' || sDept_name);
    DBMS_OUTPUT.PUT_LINE('Average Salary: ' ||TO_CHAR(nAvg_salary, 
                                                     'FML999G999G990D00',
                                                     'NLS_NUMERIC_CHARACTERS=''.,
                                                     '' NLS_CURRENCY=''R'''));
    DBMS_OUTPUT.PUT_LINE(sBorder);

    FOR emp_rec IN emp_cursor 
        LOOP
            IF emp_rec.salary > nAvg_salary THEN
                sTag := '(Above Avg)';
            ELSE
                sTag := '(Below Avg)';
            END IF;
    
            DBMS_OUTPUT.PUT_LINE(
                RPAD(emp_rec.first_name || ' ' || emp_rec.last_name, 20) ||
                RPAD(emp_rec.job_id, 12) ||
                LPAD(TO_CHAR( emp_rec.salary, 
                             'FML999G999G990',
                             'NLS_NUMERIC_CHARACTERS=''.,
                             '' NLS_CURRENCY=''R'''), 8) ||'  ' || sTag);
        END LOOP;
    
    DBMS_OUTPUT.PUT_LINE(sBorder);



EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('No department found with ID ' || proc_dept_id || '.');
        
END dept_salary_report;
/

BEGIN
    dept_salary_report(60);
END;
/

SET SERVEROUTPUT ON;
DECLARE
    iNumber_of_records    NUMBER := &number_of_records;
    sStart_id       HR.employees.employee_id%TYPE := &id;
    sCurrent_id     HR.employees.employee_id%TYPE;
    sLast_name      HR.employees.last_name%TYPE;
    
BEGIN
    DBMS_OUTPUT.PUT_LINE(iNumber_of_records || ' records requested.');
    DBMS_OUTPUT.PUT_LINE('');

    FOR i IN 0 .. iNumber_of_records - 1 
        LOOP
            sCurrent_id := sStart_id + i;
    
            SELECT last_name
                INTO   sLast_name
            FROM   HR.employees
            WHERE  employee_id = sCurrent_id;
    
            DBMS_OUTPUT.PUT_LINE(sCurrent_id || ' ' || sLast_name);
        END LOOP;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        DBMS_OUTPUT.PUT_LINE('No data was found from ID ' || sCurrent_id || '.');
END;
/
